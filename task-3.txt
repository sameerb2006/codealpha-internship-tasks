#include <stdio.h>

void add(int A[2][2], int B[2][2])
{
    int i, j;

    printf("Addition:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", A[i][j] + B[i][j]);
        }
        printf("\n");
    }
}

void multiply(int A[2][2], int B[2][2])
{
    int i, j, k, sum;

    printf("\nMultiplication:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            sum = 0;

            for(k = 0; k < 2; k++)
            {
                sum = sum + A[i][k] * B[k][j];
            }

            printf("%d ", sum);
        }
        printf("\n");
    }
}

void transpose(int A[2][2])
{
    int i, j;

    printf("\nTranspose:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", A[j][i]);
        }
        printf("\n");
    }
}

int main()
{
    int A[2][2], B[2][2];
    int i, j;

    printf("Enter Matrix A:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &A[i][j]);
        }
    }

    printf("Enter Matrix B:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &B[i][j]);
        }
    }

    add(A, B);
    multiply(A, B);
    transpose(A);

    return 0;
}
